💻 Quick Start Example

// Generate hybrid QR-SVG
const T = new Timecode({
  style: 'qr-hybrid',
  width: 1000,
  height: 1000,
  visualKey: 'secret-key'
});

const data = {
  qr_payload: "https://example.com/verify/123",
  hidden_data: {
    admin_key: "sk-abc123",
    permissions: ["read", "write"]
  }
};

const svg = await T.enc(data);

// Scan QR → https://example.com/verify/123
// Extract hidden → { admin_key: "sk-abc123", ... }

// Pattern detection agent
class PatternDetectionAgent {
  async analyze(svgString) {
    const patterns = {
      hasData: this.detectData(svgString),
      encrypted: this.detectEncryption(svgString),
      style: this.classifyStyle(svgString),
      capacity: this.estimateCapacity(svgString)
    };

    return patterns;
  }
}

// Usage

const results = await agent.analyze(svg);

// Digital Ticket System
class TicketGenerator {
  async generateTicket(info) {
    const data = {
      qr_payload: `https://verify/${info.id}`,
      ticket_id: info.id,
      event_name: info.event,
      seat: info.seat,
      purchaser: info.email,
      signature: this.sign(info)
    };

    const T = new Timecode({
      style: 'qr-hybrid',
      visualKey: SECRET_KEY
    });

    return await T.enc(data);
  }

  sign(data) {
    // HMAC signature for anti-forgery
    return crypto.createHmac('sha256', SECRET)
      .update(JSON.stringify(data))
      .digest('hex');
  }
}

// Verification at entrance
class TicketVerifier {
  async verify(svg) {
    const data = await T.dec(svg);

    if (!this.verifySignature(data)) {
      return { valid: false, reason: 'Forged' };
    }

    return { valid: true, ticket: data };
  }
}
